set -- $owned; owned="$*"
for i in $owned; do
h="${i#i-}"

echo 'instance-lookup-malformed' >&2; echo 'ERROR: instance lookup returned something that is not an instance id' >&2; exit 1

keyed="$(aws ec2 describe-instances --region 'eu-south-2' --filters 'Name=tag:forjar-key,Values=ci-box-a@eu-south-2@0123456789ab' 'Name=instance-state-name,Values=pending,running' --query 'Reservations[].Instances[].InstanceId' --output text)" || { echo 'instance-lookup-failed' >&2; echo 'ERROR: instance lookup failed (provider query exited non-zero)' >&2; exit 1; }
